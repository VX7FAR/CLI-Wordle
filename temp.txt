string answer_input()
{
    string input = "";
    while (input.length() < 5)
    {
        char c = _getch();
        if(c==0 || c==224){
            _getch();
            continue;
        }
        if (c >= 'a' && c <= 'z')
        {
            input += c;
            cout << c;
        }
        else if (c >= 'A' && c <= 'Z')
        {
            input += c;
            cout << c;
        }
        else if (c == '\b' && !input.empty())
        {
            input.pop_back();
            cout << "\b \b";
        }
        cout<<"[" << int(c)<<"]";
    }
    return input;
}
